SELECT c.region AS customers_region,
       p.product AS products_product,
       SUM(f.net_amount) AS orders_net_revenue,
       COUNT(DISTINCT f.order_id) AS orders_order_count
FROM fact_sales f
JOIN dim_customer c ON c.customer_key = f.customer_key
JOIN dim_product p ON p.product_key = f.product_key
JOIN dim_date d ON d.date_key = f.date_key
WHERE d.fiscal_quarter = @p1
GROUP BY c.region, p.product
ORDER BY customers_region ASC, products_product ASC;
